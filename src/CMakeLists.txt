macro(realcugan_add_shader SHADER_SRC)
    get_filename_component(SHADER_SRC_NAME_WE ${SHADER_SRC} NAME_WE)
    set(SHADER_COMP_HEADER ${CMAKE_CURRENT_BINARY_DIR}/s/${SHADER_SRC_NAME_WE}.comp.hex.h)

    add_custom_command(
        OUTPUT ${SHADER_COMP_HEADER}
        COMMAND ${CMAKE_COMMAND}
            -DSHADER_SRC=${CMAKE_CURRENT_SOURCE_DIR}/realcugan-ncnn-vulkan/src/${SHADER_SRC}
            -DSHADER_COMP_HEADER=${SHADER_COMP_HEADER}
            -P "${CMAKE_CURRENT_SOURCE_DIR}/realcugan-ncnn-vulkan/src/generate_shader_comp_header.cmake"
        DEPENDS
            ${CMAKE_CURRENT_SOURCE_DIR}/realcugan-ncnn-vulkan/src/${SHADER_SRC}
            ${CMAKE_CURRENT_SOURCE_DIR}/realcugan-ncnn-vulkan/src/generate_shader_comp_header.cmake
        COMMENT "Preprocessing shader source ${SHADER_SRC_NAME_WE}.comp"
        VERBATIM
    )
    set_source_files_properties(${SHADER_COMP_HEADER} PROPERTIES GENERATED TRUE)

    list(APPEND SHADER_SPV_HEX_FILES ${SHADER_COMP_HEADER})
endmacro()

realcugan_add_shader(realcugan_preproc.comp)

# Emit deterministic cross-platform link manifest for build.rs
set(_link_manifest_lines
    "LIB_SUBDIR=${CMAKE_INSTALL_LIBDIR}"
    "STATIC_LIB=realcugan-ncnn-vulkan-wrapper"
    "STATIC_LIB=ncnn"
)

if (TARGET SPIRV)
    list(APPEND _link_manifest_lines "STATIC_LIB=SPIRV")
endif ()
if (TARGET glslang)
    list(APPEND _link_manifest_lines "STATIC_LIB=glslang")
endif ()
if (TARGET MachineIndependent)
    list(APPEND _link_manifest_lines "STATIC_LIB=MachineIndependent")
endif ()
if (TARGET GenericCodeGen)
    list(APPEND _link_manifest_lines "STATIC_LIB=GenericCodeGen")
endif ()

if (OpenMP_CXX_FOUND)
    if (OpenMP_CXX_LIBRARIES)
        foreach (_omp_lib IN LISTS OpenMP_CXX_LIBRARIES)
            if (EXISTS "${_omp_lib}")
                get_filename_component(_omp_dir "${_omp_lib}" DIRECTORY)
                get_filename_component(_omp_name "${_omp_lib}" NAME_WE)
                string(REGEX REPLACE "^lib" "" _omp_name "${_omp_name}")
                list(APPEND _link_manifest_lines "SEARCH_DIR=${_omp_dir}")
                list(APPEND _link_manifest_lines "DYLIB=${_omp_name}")
            elseif (_omp_lib MATCHES "^-l(.+)$")
                list(APPEND _link_manifest_lines "DYLIB=${CMAKE_MATCH_1}")
            elseif (_omp_lib MATCHES "^-")
                list(APPEND _link_manifest_lines "LINK_ARG=${_omp_lib}")
            elseif (_omp_lib)
                list(APPEND _link_manifest_lines "DYLIB=${_omp_lib}")
            endif ()
        endforeach ()
    elseif (MSVC)
        list(APPEND _link_manifest_lines "DYLIB=vcomp")
    endif ()
endif ()

if (UNIX AND NOT APPLE)
    if (CMAKE_DL_LIBS)
        foreach (_dl_lib IN LISTS CMAKE_DL_LIBS)
            list(APPEND _link_manifest_lines "DYLIB=${_dl_lib}")
        endforeach ()
    endif ()
    if (CMAKE_THREAD_LIBS_INIT)
        if (CMAKE_THREAD_LIBS_INIT MATCHES "pthread")
            list(APPEND _link_manifest_lines "DYLIB=pthread")
        endif ()
    endif ()
endif ()

if (APPLE AND NCNN_VULKAN AND NCNN_SIMPLEVK)
    if (DEFINED Vulkan_LIBRARY AND EXISTS "${Vulkan_LIBRARY}")
        get_filename_component(_vk_dir "${Vulkan_LIBRARY}" DIRECTORY)
        get_filename_component(_vk_name "${Vulkan_LIBRARY}" NAME_WE)
        string(REGEX REPLACE "^lib" "" _vk_name "${_vk_name}")
        list(APPEND _link_manifest_lines "SEARCH_DIR=${_vk_dir}")
        list(APPEND _link_manifest_lines "DYLIB=${_vk_name}")
        foreach (_fw IN ITEMS Metal Foundation QuartzCore CoreGraphics IOSurface AppKit IOKit)
            list(APPEND _link_manifest_lines "FRAMEWORK=${_fw}")
        endforeach ()
    else ()
        set(_simplevk_tbd "${CMAKE_CURRENT_SOURCE_DIR}/realcugan-ncnn-vulkan/src/ncnn/src/simplevk.tbd")
        if (EXISTS "${_simplevk_tbd}")
            list(APPEND _link_manifest_lines "LINK_ARG=-Wl,-weak_library,${_simplevk_tbd}")
        endif ()
    endif ()
endif ()

list(REMOVE_DUPLICATES _link_manifest_lines)
string(JOIN "\n" _link_manifest_content ${_link_manifest_lines})
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/link_manifest.txt" "${_link_manifest_content}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/link_manifest.txt" DESTINATION ".")
